cmake_minimum_required(VERSION 3.22)
project(stm_prawler_driver_tests C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

option(ENABLE_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
add_executable(test_li8x0 test_li8x0.c ../Core/Src/LI8x0.c)
target_include_directories(test_li8x0 PRIVATE stubs ../Core/Inc)
target_compile_options(test_li8x0 PRIVATE -Wall -Wextra -Werror -pedantic -UNDEBUG)
target_link_libraries(test_li8x0 PRIVATE m)
if(ENABLE_SANITIZERS)
    target_compile_options(test_li8x0 PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(test_li8x0 PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME li8x0 COMMAND test_li8x0)
